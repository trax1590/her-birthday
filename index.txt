<script>
// 🎂 BIRTHDAY DATE
const birthday = new Date("Jan 01, 2026 00:00:00").getTime();

const btn = document.getElementById("surpriseBtn");
const msg = document.getElementById("surpriseMessage");

setInterval(() => {
  const now = new Date().getTime();
  const diff = birthday - now;

  if (diff <= 0) {
    btn.disabled = false;
    btn.style.cursor = "pointer";
    btn.innerText = "🎉 Open Your Note 💝";
  }

  document.getElementById("days").innerText =
    Math.max(0, Math.floor(diff / (1000 * 60 * 60 * 24)));
  document.getElementById("hours").innerText =
    Math.max(0, Math.floor((diff / (1000 * 60 * 60)) % 24));
  document.getElementById("minutes").innerText =
    Math.max(0, Math.floor((diff / (1000 * 60)) % 60));
  document.getElementById("seconds").innerText =
    Math.max(0, Math.floor((diff / 1000) % 60));
}, 1000);

// 🎁 Surprise click
btn.onclick = () => {
  msg.style.display = "block";

  for (let i = 0; i < 60; i++) {
    const heart = document.createElement("div");
    heart.innerText = "🎆💖🎉";
    heart.style.position = "fixed";
    heart.style.left = Math.random() * 100 + "vw";
    heart.style.top = "100vh";
    heart.style.fontSize = "24px";
    heart.style.animation = "floatUp 4s linear";
    document.body.appendChild(heart);
    setTimeout(() => heart.remove(), 4000);
  }
};

// 🖼️ Slideshow
let index = 0;
const images = document.querySelectorAll(".gallery img");
setInterval(() => {
  images[index].classList.remove("active");
  index = (index + 1) % images.length;
  images[index].classList.add("active");
}, 3000);

// 💖 Floating hearts always
setInterval(() => {
  const heart = document.createElement("div");
  heart.className = "heart";
  heart.innerText = "💖";
  heart.style.left = Math.random() * 100 + "vw";
  heart.style.animationDuration = (4 + Math.random() * 4) + "s";
  document.body.appendChild(heart);
  setTimeout(() => heart.remove(), 7000);
}, 600);
</script>
